\begin{frame}{Сначала амплитуды. Потом вероятность}
\noindent{\scriptsize Релятивистская запись: $p^\mu=(E,\mathbf p)$, $q^\mu=p_1^\mu-p_2^\mu$, $K^\mu=(p_1^\mu+p_2^\mu)/2$; $c=1$.}\par
\begin{columns}[T,onlytextwidth]
\column{.73\textwidth}
\[a=e^{-i(p_1\cdot x_A+p_2\cdot x_B)/\hc},\qquad
 b=e^{-i(p_2\cdot x_A+p_1\cdot x_B)/\hc}.\]
\[\Psi=\frac{a+b}{\sqrt2},\qquad
 |\Psi|^2=\frac{|a|^2+|b|^2+2\operatorname{Re}(ab^*)}{2}.\]
\[\boxed{|\Psi|^2=1+\cos\!\left(\frac{q\cdot\Delta x}{\hc}\right)}\]
\[q\cdot\Delta x=q^0\Delta(ct)-\mathbf q\cdot\Delta\mathbf r.\]
{\small $p$ — ГэВ; $x$ — фм; $\hc\simeq0.197$ ГэВ·фм.\par}
\column{.23\textwidth}
\tagline{НАПОМИНАНИЕ}
$P\propto |a|^2$\par вероятность.\par\vspace{2mm}
$e^{i\varphi}$\par волна с фазой $\varphi$.\par\vspace{2mm}
$*$ — сопряжение;\par $\operatorname{Re}$ —\par действительная часть.
\end{columns}
\takeaway{При $q\to0$ идеальный обменный вклад даёт двукратное усиление.}
\src{Плоские волны, близкие импульсы, одинаковые модули амплитуд; FSI пока нет. $\Delta x=x_A-x_B$, $x=(ct,\mathbf r)$ в фм, $p$ в ГэВ; $\hc=0.19732698$ ГэВ·фм.}

\begin{frame}{Средний вес имеет другую статистическую ошибку}
Для $n$ одних и тех же модельных пар:
\[C=\frac{\sum_iw_i}{n}=\overline w,\qquad
\widehat{\operatorname{Var}}(\overline w)=\frac{\sum_i(w_i-\overline w)^2}{n(n-1)}.\]
\small Условная оценка для независимых пар при фиксированном $n>1$.\par
Общие треки требуют анализа на уровне событий.\par\vspace{3mm}
Для отношения $C=\mathcal NA/B$ с фиксированной $\mathcal N$:
\[\frac{\operatorname{Var}(C)}{C^2}\simeq
\frac{\operatorname{Var}(A)}{A^2}+\frac{\operatorname{Var}(B)}{B^2}
-2\frac{\operatorname{Cov}(A,B)}{AB}.\]
\takeaway{Ковариация существенна, если числитель и знаменатель используют одни пары.}
\src{Вывод через дисперсию среднего и линейное распространение ошибок. При оценке нормировки по данным нужны также её дисперсия и ковариации.}
